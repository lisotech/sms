POST /v1/sms/send (PHP cURL Example) 
$curl = curl_init();

curl_setopt_array($curl, [
  CURLOPT_URL => "https://api.zambiabulksms.zm/v1/sms/send",
  CURLOPT_RETURNTRANSFER => true,
  CURLOPT_POST => true,
  CURLOPT_POSTFIELDS => json_encode([
    "sender_id" => "ZambiaAlert",
    "recipients" => ["0977123456", "0966987654", "0955112233"],
    "message" => "Your account balance is K450.00."
  ]),
  CURLOPT_HTTPHEADER => [
    "Authorization: Bearer YOUR_API_KEY",
    "Content-Type: application/json"
  ],
]);

$response = curl_exec($curl);
curl_close($curl);
echo $response;
